## Step 1 — capture a live getcustomers request (URL + headers + tokens)
Run this with browserEval to install an interceptor:
```js

const X = window.XMLHttpRequest;
window.XMLHttpRequest = function(){
  const x = new X(); let u='', hdrs={};
  const o = x.open; x.open = function(m, uu){ u = uu; return o.apply(x, arguments); };
  const sh = x.setRequestHeader; x.setRequestHeader = function(h, v){ hdrs[h] = v; return sh.apply(x, arguments); };
  const s = x.send; x.send = function(b){
    if (/getcustomers/i.test(String(u))) x.addEventListener('load', () => { window.__cap = { url: u, headers: hdrs, body: b }; });
    return s.apply(x, arguments);
  };
  return x;
};
'installed';
```

## Step 2 — replay every page
Once window.__cap is set, run ONE browserEval that paginates through all records:
```js
const cap = window.__cap;
const base = JSON.parse(cap.body);
const byId = new Map();           // dedupe by UserID as we go — pages can overlap
let page = 1, total = Infinity, emptyRetries = 0;
const MAX_PAGES = 300;            // hard safety cap so a bad response can't loop forever
while (byId.size < total && page <= MAX_PAGES) {
  const body = JSON.stringify({ ...base, PageIndex: page, PageSize: 200 });
  let arr = [];
  try {
    const r = await fetch(cap.url, { method: 'POST', headers: { ...cap.headers, 'Content-Type': 'application/json' }, body });
    const j = await r.json();
    arr = (function find(o){ for (const k in o){ if (Array.isArray(o[k]) && o[k].length) return o[k]; if (o[k] && typeof o[k]==='object'){ const f = find(o[k]); if (f) return f; } } return []; })(j);
  } catch (e) { arr = []; }
  if (!arr.length) {
    // A transient empty/errored page must NOT end pagination early — retry the same page once.
    if (emptyRetries < 1) { emptyRetries++; await new Promise(res => setTimeout(res, 800)); continue; }
    break; // retried and still empty → genuinely no more rows
  }
  emptyRetries = 0;
  total = arr[0].TotalRecords ?? total;
  for (const c of arr) byId.set(String(c.UserID), c);
  page++;
}
window.__all = [...byId.values()];
return { collected: window.__all.length, total, pages: page - 1 };
```
Confirm `collected` reaches `total` (the TotalRecords for this account). If not, fix and retry before
finishing.

## Step 3 — map the records and DOWNLOAD them as a file

be thousands and they will be truncated. Instead, build the mapped list in the page and trigger a file
download. The download is retrieved automatically (Browserbase Session Downloads). Run this browserEval:
```js
const all = window.__all || [];
const j = (v) => (v == null ? "" : String(v));
const mapped = all.map(c => ({
  id: j(c.UserID),
  name: (j(c.FirstName) + " " + j(c.LastName)).trim(),
  email: j(c.EmailAddress),
  mobile: j(c.CellPhone),
  dayPhone: j(c.DayPhone),
  nightPhone: j(c.NightPhone),
  address: [c.StreetAddress, c.City, c.State, c.Zip].map(j).filter(Boolean).join(", "),
  birthday: j(c.strBirthDate),
  gender: j(c.Gender),
  customerSince: j(c.strCustomerSince),
  lastVisit: j(c.strLastVisited),
  membership: j(c.MembershipName),
  referredBy: j(c.ReferredBy),
  tags: j(c.GeneralTag),
}));
// dedupe by id
const seen = new Set(); const out = mapped.filter(c => !seen.has(c.id) && seen.add(c.id));
const blob = new Blob([JSON.stringify(out, null, 2)], { type: "application/json" });
```
